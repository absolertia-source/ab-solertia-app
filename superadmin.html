window.crearEmpresa = async () => { const nombre = document.getElementById('empNombre').value.trim(); const idEmp = document.getElementById('empId').value.trim(); const aEmail = document.getElementById('adminEmail').value.trim(); const aPass = document.getElementById('adminPass').value.trim(); if(!nombre || !idEmp || !aEmail || !aPass){ alert('Completa todos los campos'); return; } if(aPass.length < 6){ alert('La contraseña debe tener al menos 6 caracteres'); return; } if(!/^[a-z0-9\-]+$/.test(idEmp)){ alert('El ID solo puede tener letras minúsculas, números y guiones. Ej: distribuidora-norte'); return; } let paso = ''; try { // ----- PASO 1: Crear empresa ----- paso = '1/3 · Crear empresa'; console.log('▶️ ' + paso); const empRef = doc(db, "empresas", idEmp); const empSnap = await getDoc(empRef); if(empSnap.exists()){ alert('⚠️ Ya existe una empresa con ese ID. Usa otro.'); return; } await setDoc(empRef, { empresa_nombre: nombre, activa: true, creada: Timestamp.now() }); console.log('✅ Empresa creada'); // ----- PASO 2: Crear usuario en Auth ----- paso = '2/3 · Crear usuario en Authentication'; console.log('▶️ ' + paso); const cred = await createUserWithEmailAndPassword(secondaryAuth, aEmail, aPass); console.log('✅ Auth user creado con UID: ' + cred.user.uid); // ----- PASO 3: Crear perfil en Firestore ----- paso = '3/3 · Guardar perfil en /usuarios'; console.log('▶️ ' + paso); await setDoc(doc(db, "usuarios", cred.user.uid), { email: aEmail, rol: "admin", empresa_id: idEmp, empresa_nombre: nombre, creado: Timestamp.now() }); console.log('✅ Perfil guardado en Firestore'); await signOut(secondaryAuth); alert(`✅ Todo OK\n\nEmpresa: ${nombre}\nAdmin: ${aEmail}\n\nEl admin ya puede entrar en admin.html con:\n• Correo: ${aEmail}\n• Contraseña: la que pusiste`); document.getElementById('empNombre').value = ''; document.getElementById('empId').value = ''; document.getElementById('adminEmail').value = ''; document.getElementById('adminPass').value = ''; cargarGlobal(); } catch(e) { console.error('❌ Error en ' + paso, e); if(e.code === 'auth/email-already-in-use'){ alert(`❌ FALLÓ en ${paso}\n\nEse correo ya está registrado en Authentication.\n\n` + `Puede ser un "usuario huérfano": existe en Auth pero sin perfil en Firestore.\n\n` + `Soluciones:\n` + `1) Ve a Firebase Console → Authentication → Users, borra ese correo y vuelve a intentar.\n` + `2) O crea manualmente su perfil en Firestore → /usuarios/{UID} con rol "admin".`); } else if(e.code === 'permission-denied' || (e.message && e.message.includes('permission'))) { alert(`❌ FALLÓ en ${paso} (permisos de Firestore)\n\n` + `Las reglas de Firestore bloquearon esta operación.\n` + `Verifica que las reglas publicadas permitan al superadmin crear usuarios con rol "admin".`); } else { alert(`❌ FALLÓ en ${paso}\n\nCódigo: ${e.code || '(sin código)'}\nMensaje: ${e.message}`); } } }
